## Design

Based on inspection of `http_lifecycle.py`:
```python
# Current (no validation):
proc = subprocess.Popen(  # nosec B603
    cfg.cmd,
    stdout=subprocess.DEVNULL,
    stderr=stderr_fh,
    env=env,
    start_new_session=True,
)

# Proposed fix:
_ALLOWED_COMMANDS = frozenset({"node", "npm", "npx"})  # Adjust based on production needs

def _validate_cmd(cmd: str) -> None:
    """Validate that the command is in the allowed list."""
    base_cmd = Path(cmd.split()[0]).name if cmd else ""
    if base_cmd not in _ALLOWED_COMMANDS:
        raise ValueError(f"Command '{base_cmd}' is not in the allowed list")

_validate_cmd(cfg.cmd)
proc = subprocess.Popen(  # nosec B603
    cfg.cmd,
    stdout=subprocess.DEVNULL,
    stderr=stderr_fh,
    env=env,
    start_new_session=True,
)
```
