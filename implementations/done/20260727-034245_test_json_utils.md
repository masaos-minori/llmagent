## Design

Based on inspection of `json_utils.py`:
```python
# extract_llm_content has four validation branches:
# 1. Missing or empty 'choices'
# 2. choices[0] is not a dict
# 3. choices[0].message is not a dict
# 4. content is not a str

# Other functions:
# - dumps(): deterministic JSON serialization with sorted keys
# - now_iso()/now_iso_raw(): ISO timestamp generation
# - tool_call_serialized_length()/serialized_length(): length calculation
# - parse_http_json(): HTTP response parsing
```

The test will verify all validation branches and normal operation paths.

   ```python
   """Characterization tests for json_utils."""
   
   import pytest
   from shared.json_utils import (
       dumps,
       extract_llm_content,
       now_iso,
       now_iso_raw,
       parse_http_json,
       serialized_length,
       tool_call_serialized_length,
   )
   
   # --- extract_llm_content ---
   
   def test_extract_llm_content_valid():
       data = {"choices": [{"message": {"content": "Hello"}}]}
       assert extract_llm_content(data) == "Hello"
   
   def test_extract_llm_content_strips_whitespace():
       data = {"choices": [{"message": {"content": "  Hello  "}}]}
       assert extract_llm_content(data) == "Hello"
   
   def test_extract_llm_content_missing_choices():
       data = {}
       with pytest.raises(ValueError, match="Missing or empty"):
           extract_llm_content(data)
   
   def test_extract_llm_content_empty_choices():
       data = {"choices": []}
       with pytest.raises(ValueError, match="Missing or empty"):
           extract_llm_content(data)
   
   def test_extract_llm_content_non_list_choices():
       data = {"choices": "not a list"}
       with pytest.raises(ValueError, match="Missing or empty"):
           extract_llm_content(data)
   
   def test_extract_llm_content_non_dict_first_choice():
       data = {"choices": ["not a dict"]}
       with pytest.raises(ValueError, match="choices\[0\] is not a dict"):
           extract_llm_content(data)
   
   def test_extract_llm_content_missing_message():
       data = {"choices": [{}]}
       with pytest.raises(ValueError, match="choices\[0\].message is not a dict"):
           extract_llm_content(data)
   
   def test_extract_llm_content_non_dict_message():
       data = {"choices": [{"message": "not a dict"}]}
       with pytest.raises(ValueError, match="choices\[0\].message is not a dict"):
           extract_llm_content(data)
   
   def test_extract_llm_content_non_string_content():
       data = {"choices": [{"message": {"content": 123}}]}
       with pytest.raises(ValueError, match="content is not a str"):
           extract_llm_content(data)
   
   # --- dumps ---
   
   def test_dumps_deterministic():
       data = {"b": 2, "a": 1}
       result = dumps(data)
       assert '"a": 1' in result
       assert '"b": 2' in result
   
   # --- now_iso / now_iso_raw ---
   
   def test_now_iso_format():
       result = now_iso()
       assert result.endswith("+00:00")
   
   def test_now_iso_raw_format():
       result = now_iso_raw()
       assert result.endswith("+00:00")
   
   # --- parse_http_json ---
   
   def test_parse_http_json_valid(mocker):
       resp = mocker.Mock()
       resp.status_code = 200
       resp.content = b'{"key": "value"}'
       resp.headers = {"content-type": "application/json"}
       result = parse_http_json(resp)
       assert result == {"key": "value"}
   
   def test_parse_http_json_invalid(mocker):
       resp = mocker.Mock()
       resp.status_code = 200
       resp.content = b"not json"
       resp.headers = {"content-type": "application/json"}
       with pytest.raises(Exception):
           parse_http_json(resp)
   ```
