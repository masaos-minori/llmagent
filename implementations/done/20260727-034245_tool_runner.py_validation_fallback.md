```python
        runtime_tool = registry.get(name)

        return None
    # ... validate using runtime_tool.input_schema

# Proposed (with gateway fallback):
def _validate_tool_call(ctx, name, args):
    registry = ctx.services_required.runtime_tools
    if registry is not None:
        try:
            runtime_tool = registry.get(name)
            result = validate_tool_arguments(...)
            if not result.success:
                return ToolCallResult(...)
            return None
        except KeyError:
            pass
    
    # Fallback: check gateway's tool definitions
    if hasattr(ctx.services_required, 'gateway') and ctx.services_required.gateway is not None:
        # Query gateway or cfg.tool.tool_definitions for schema
        # ... implement fallback validation
```

Note: The exact fallback mechanism depends on whether the gateway exposes tool definitions or if we need to query `ctx.cfg.tool.tool_definitions`.
