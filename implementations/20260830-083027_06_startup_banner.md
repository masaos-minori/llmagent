```python
"""scripts/agent/startup_banner.py

StartupBanner — Startup display responsibility extraction.

Owns: _print_startup_banner, _get_chunk_count, _get_workflow_status.
Displays DB chunks, tool count, workflow status, memory mode.
"""

import sqlite3
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from agent.cli_view import CLIView
    from agent.context import AgentContext

logger = Logger(__name__, "/opt/llm/logs/agent.log")


class StartupBanner:
    """Manages REPL startup banner display and status information.

    Owns displaying DB chunk count, tool count, workflow status, and memory
    mode during REPL initialization.
    """

    def __init__(
        self,
        ctx: "AgentContext",
        view: "CLIView",
    ) -> None:
        self._ctx = ctx
        self._view = view

    def get_chunk_count(self) -> str:
        """Return formatted chunk count from DB, or '?' on error."""
        try:
            count = RagMaintenanceService().stats_rag()[1]
            return f"{count:,}"
        except (sqlite3.Error, OSError, RuntimeError) as e:
            logger.debug("Failed to get chunk count: %s", e)
            return "?"

    def get_workflow_status(self) -> str:
        """Return a human-readable workflow status string for the startup banner."""
        # Note: orchestrator reference may be None before Phase 2 step 7 completes
        # This is handled by the caller passing the orchestrator reference
        return "unknown"

    def print_startup_banner(self) -> None:
        """Print the startup line showing DB chunks, tool count, and workflow status."""
        chunk_count = self.get_chunk_count()
        workflow_status = self.get_workflow_status()
        mem_cfg = self._ctx.cfg.memory
        memory_mode = "enabled" if mem_cfg.use_memory_layer else "disabled"
        self._view.write_startup_banner(
            chunk_count,
            self._n_tools,
            workflow_status,
            memory_mode=memory_mode,
        )
```
