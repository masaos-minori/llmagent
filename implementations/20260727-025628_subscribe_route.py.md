# Current (broken):
from typing import Any

async def _sse_gen() -> Any:
    sub = broker.subscribe(list(topic))
    try:
        rows = await run_with_db_lock(_fetch_replay)

# Proposed fix:

async def _sse_gen() -> AsyncGenerator[str, None]:

        replay_ceil = start_seq   # move BEFORE for loop — always initialized
        for row in rows:
            ...
            replay_ceil = row["seq"]
