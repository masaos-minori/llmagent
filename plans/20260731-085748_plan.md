## Goal

## Design

### Approach: Use optimistic locking via conditional UPDATE

Since SQLite doesn't support `FOR UPDATE SKIP LOCKED`, we'll use optimistic locking:

```python
def recover_stale_attempts(self, db: SQLiteHelper) -> int:
    """Recover stale workflow attempts with optimistic locking.
    
    Returns the number of attempts successfully recovered.
    """
    grace_period = timedelta(seconds=30)
    stale_threshold = datetime.now() - grace_period
    
    # Get list of stale attempts (without locking)
    cursor = db.execute("""
        SELECT id, workflow_id FROM workflow_attempts
        WHERE status = 'running'
        AND updated_at < ?
        ORDER BY created_at ASC
    """, (stale_threshold.isoformat(),))
    
    stale_rows = cursor.fetchall()
    recovered_count = 0
    
    for row in stale_rows:
        # Try to claim this attempt atomically
        update_cursor = db.execute("""
            UPDATE workflow_attempts
            SET status = 'recovered', updated_at = ?
            WHERE id = ? AND status = 'running'
        """, (datetime.now().isoformat(), row['id']))
        
        # Check if we claimed it (rows affected > 0 means we got it)
        if update_cursor.rowcount > 0:
            logger.info(f"Recovered stale attempt {row['id']}")
            self._recover_attempt(row['id'])
            db.commit()
            recovered_count += 1
        else:
            # Another process claimed it first
            logger.debug(f"Attempt {row['id']} was already claimed by another process")
    
    return recovered_count
```
