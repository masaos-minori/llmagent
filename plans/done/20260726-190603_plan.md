## Design

Create guard tests:
```python
import pytest
from shared.json_utils import extract_llm_content, dumps, now_iso

def test_extract_llm_content_valid():
    data = {"choices": [{"message": {"content": "Hello world"}}]}
    assert extract_llm_content(data) == "Hello world"

def test_extract_llm_content_missing_choices():
    with pytest.raises(ValueError, match="Missing or empty"):
        extract_llm_content({})

def test_extract_llm_content_empty_choices():
    with pytest.raises(ValueError, match="Missing or empty"):
        extract_llm_content({"choices": []})

def test_extract_llm_content_invalid_first_choice():
    with pytest.raises(ValueError, match="not a dict"):
        extract_llm_content({"choices": ["invalid"]})

def test_extract_llm_content_invalid_message():
    with pytest.raises(ValueError, match="not a dict"):
        extract_llm_content({"choices": [{"message": "invalid"}]})

def test_extract_llm_content_invalid_content_type():
    with pytest.raises(ValueError, match="not a str"):
        extract_llm_content({"choices": [{"message": {"content": 123}}]})

def test_dumps_returns_str():
    result = dumps({"key": "value"})
    assert isinstance(result, str)

def test_now_iso_format():
    result = now_iso()
    assert result.endswith("Z")
```
