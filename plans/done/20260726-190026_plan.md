## Design

Add whitelist and validation:
```python
_ALLOWED_MCP_COMMANDS = frozenset({
    "uvicorn",
    "fastmcp",
    "python",
    "python3",
})

def start(...) -> None:
    ...
    if cfg.cmd:
        cmd_base = os.path.basename(cfg.cmd[0])
        if cmd_base not in _ALLOWED_MCP_COMMANDS:
            logger.error(
                "MCP subprocess rejected: %r not in allowed commands %s",
                cfg.cmd[0],
                _ALLOWED_MCP_COMMANDS,
            )
            raise HttpStartupError(
                StartupFailure(server_key=server_key, reason="unauthorized command")
            )
    
    proc = subprocess.Popen(
        cfg.cmd,
        ...
    )
```
