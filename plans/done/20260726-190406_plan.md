```python
import pytest
from shared.llm_hot_config import LlmHotConfigHandler

class MockLLMClient:
    temperature = 0.5
    max_tokens = 100
    max_retries = 3
    retry_base_delay = 1.0
    sse_heartbeat_timeout = 30.0
    sse_malformed_retry = 3
    sse_reconnect_max = 5
    stream_retry_on_heartbeat_timeout = True
    stream_retry_on_malformed_chunk = True

@pytest.mark.parametrize("field,kwarg,value", LlmHotConfigHandler.HOT_CONFIG_FIELDS)
def test_apply_one_sets_attribute(field, kwarg, value):
    obj = MockLLMClient()
    LlmHotConfigHandler.apply_one(obj, field, kwarg, value)
    assert getattr(obj, field) == value

def test_apply_config_updates_all_fields():
    obj = MockLLMClient()
    LlmHotConfigHandler.apply_config(
        obj,
        temperature=0.7,
        max_tokens=200,
        max_retries=5,
        retry_base_delay=2.0,
        sse_heartbeat_timeout=60.0,
        sse_malformed_retry=5,
        sse_reconnect_max=10,
        stream_retry_on_heartbeat_timeout=False,
        stream_retry_on_malformed_chunk=False,
    )
    assert obj.temperature == 0.7
    assert obj.max_tokens == 200
    assert obj.max_retries == 5
    assert obj.retry_base_delay == 2.0
    assert obj.sse_heartbeat_timeout == 60.0
    assert obj.sse_malformed_retry == 5
    assert obj.sse_reconnect_max == 10
    assert obj.stream_retry_on_heartbeat_timeout is False
    assert obj.stream_retry_on_malformed_chunk is False

def test_apply_config_partial_update():
    obj = MockLLMClient()
    original_temp = obj.temperature
    LlmHotConfigHandler.apply_config(obj, temperature=0.7)
    assert obj.temperature == 0.7
    assert obj.max_tokens == original_temp  # unchanged
```
