## Design

Add four new methods to StateStore for diagnostic queries:
- `count_tasks_by_session(session_id)` — returns task count
- `count_workflows_by_session(session_id)` — returns distinct workflow count
- `count_approvals_by_session(session_id)` — returns approval event count
- `count_retries_by_session(session_id)` — returns execute-stage retry count
- `list_artifacts_by_session(session_id)` — returns artifact URIs

Replace the direct SQLiteHelper connection in `_persist_session_diagnostics()` with calls to these new StateStore methods.

Current StateStore has task-related methods (`get_task_by_id`, `get_task_by_idempotency_key`, `get_task_by_session`, `get_latest_task`, `list_tasks`) but lacks the aggregate/diagnostic queries needed by `_persist_session_diagnostics()`.

Add after existing Task section methods:

```python
# ── Diagnostics ───────────────────────────────────────────────────────────────

def count_tasks_by_session(self, session_id: str) -> int:
    """Return the number of tasks for a given session."""
    rows = self._db.fetchall(
        "SELECT COUNT(*) as cnt FROM tasks WHERE session_id=?",
        (session_id,),
    )
    return int(rows[0]["cnt"]) if rows else 0

def count_workflows_by_session(self, session_id: str) -> int:
    """Return the number of distinct workflows for a given session."""
    rows = self._db.fetchall(
        "SELECT COUNT(DISTINCT workflow_id) as cnt"
        " FROM tasks WHERE session_id=? AND workflow_id IS NOT NULL",
        (session_id,),
    )
    return int(rows[0]["cnt"]) if rows else 0

def count_approvals_by_session(self, session_id: str) -> int:
    """Return the number of approval events for a given session."""
    rows = self._db.fetchall(
        "SELECT COUNT(*) as cnt FROM approvals"
        " WHERE task_id IN (SELECT task_id FROM tasks WHERE session_id=?)",
        (session_id,),
    )
    return int(rows[0]["cnt"]) if rows else 0

def count_retries_by_session(self, session_id: str) -> int:
    """Return the number of execute-stage retries for a given session.

    Retries are counted as attempts where stage_id='execute' minus the number
    of tasks (since each task gets exactly one initial attempt).
    """
    rows = self._db.fetchall(
        "SELECT COUNT(*) as cnt FROM attempts"
        " WHERE task_id IN (SELECT task_id FROM tasks WHERE session_id=?)"
        " AND stage_id='execute'",
        (session_id,),
    )
    task_count = self.count_tasks_by_session(session_id)
    raw_attempts = int(rows[0]["cnt"]) if rows else 0
    return max(0, raw_attempts - task_count)

def list_artifacts_by_session(self, session_id: str) -> list[str]:
    """Return artifact URIs for a given session."""
    rows = self._db.fetchall(
        "SELECT uri FROM artifacts WHERE task_id IN (SELECT task_id FROM tasks WHERE session_id=?)",
        (session_id,),
    )
    return [dict(r)["uri"] for r in rows]
```
